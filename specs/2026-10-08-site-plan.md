# Hypospray Site Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the `docs/` Starlight site into `site/`: an LCARS-style landing page at `/` plus the Starlight docs at `/docs/…`, sharing tokens, fonts, code theme and header components.

**Architecture:** One Astro 7 project. The landing page is a plain Astro page built from `.astro` components (`components/lcars/*` shared with the docs, `components/landing/*` landing-only), with small per-component `<script>`s for behaviour. Starlight renders `/docs/…` from `src/content/docs/docs/`, themed through `customCss` and component overrides (`Header`, `Head`, `ThemeProvider`, `ThemeSelect`, `MobileMenuFooter`) that reuse the LCARS components. All code samples are real `.ts` files rendered through Expressive Code.

**Tech Stack:** Astro 7.2, Starlight 0.41, Expressive Code (bundled with Starlight), Astro Fonts API (Fontsource provider), TypeScript, vitest 4, pnpm workspaces, jj (Jujutsu) for version control.

**Spec:** `specs/2026-10-08-site-design.md`. Visual numbers: `specs/2026-10-05-landing-page-visual-design.md`. Reference renders (gitignored, local only): `specs/assets/landing-page/Hypospray — Final Designs.pdf`.

- [ ] **Step 4: Implement `region.ts`**

`site/src/lib/region.ts`:

```ts
const MARKER = /^\s*\/\/\s*#(end)?region\b\s*(\S*)\s*$/;

function isMarker(line: string): boolean {
  return MARKER.test(line);
}

function dedent(lines: string[]): string[] {
  const indents = lines
    .filter((line) => line.trim() !== '')
    .map((line) => line.length - line.trimStart().length);
  const min = indents.length > 0 ? Math.min(...indents) : 0;
  return lines.map((line) => line.slice(min));
}

function trimBlankEdges(lines: string[]): string[] {
  let start = 0;
  let end = lines.length;
  while (start < end && lines[start].trim() === '') start++;
  while (end > start && lines[end - 1].trim() === '') end--;
  return lines.slice(start, end);
}

/**
 * Returns the lines between `// #region name` and its matching `// #endregion`,
 * dedented, with every region marker removed. Without a name, returns the whole
 * source with markers removed. Throws if the region is missing or unclosed.
 */
export function extractRegion(source: string, name?: string): string {
  const lines = source.split('\n');

  if (name === undefined) {
    return trimBlankEdges(lines.filter((line) => !isMarker(line))).join('\n');
  }

  const start = lines.findIndex((line) => {
    const match = MARKER.exec(line);
    return match !== null && match[1] === undefined && match[2] === name;
  });
  if (start === -1) throw new Error(`Region "${name}" not found`);

  const body: string[] = [];
  let depth = 1;
  for (const line of lines.slice(start + 1)) {
    const match = MARKER.exec(line);
    if (match) {
      depth += match[1] === undefined ? 1 : -1;
      if (depth === 0) return dedent(trimBlankEdges(body)).join('\n');
      continue;
    }
    body.push(line);
  }
  throw new Error(`Region "${name}" is not closed`);
}
```

`site/src/lib/stardate.ts`:

```ts
/**
 * A TNG-flavoured stardate: TNG premiered in 1987 at stardate 41xxx and
 * advanced roughly 1000 per season, so each real year adds 1000.
 */
export function stardate(date: Date): string {
  const year = date.getUTCFullYear();
  const start = Date.UTC(year, 0, 1);
  const end = Date.UTC(year + 1, 0, 1);
  const fraction = (date.getTime() - start) / (end - start);
  const value = 41000 + 1000 * (year - 1987) + 1000 * fraction;
  return value.toFixed(1);
}
```

`site/src/lib/url.ts`:

```ts
const EXTERNAL = /^(?:[a-z][a-z\d+.-]*:|\/\/|#)/i;

/** Prefixes an internal path with the site's base path. External URLs and bare anchors pass through. */
export function withBase(path: string, base: string = import.meta.env.BASE_URL): string {
  if (EXTERNAL.test(path)) return path;
  const prefix = base.endsWith('/') ? base : `${base}/`;
  return prefix + path.replace(/^\//, '');
}
```

```astro
{!isStatic && (
  <script>
    const header = document.querySelector<HTMLElement>('[data-lcars-header]:not([data-static])');
    const sentinel = document.querySelector('[data-header-sentinel]');
    const sections = [...document.querySelectorAll<HTMLElement>('main section[id]')];
    const navLinks = [...document.querySelectorAll<HTMLAnchorElement>('[data-nav-section]')];
    const phone = window.matchMedia('(max-width: 599.98px)');
    const reduced = window.matchMedia('(prefers-reduced-motion: reduce)');

    if (header && sentinel) {
      new IntersectionObserver(([entry]) => {
        header.toggleAttribute('data-compact', !entry.isIntersecting && entry.boundingClientRect.top < 0);
      }).observe(sentinel);
    }

    let suppressed = false;
    let releaseTimer: number | undefined;

    function select(id: string | null) {
      for (const link of navLinks) {
        if (id !== null && link.dataset.navSection === id) link.setAttribute('aria-current', 'location');
        else link.removeAttribute('aria-current');
      }
    }

    // A 1px detection line just under the compact header; the section crossing it is active.
    let sectionObserver: IntersectionObserver | undefined;
    function observeSections() {
      sectionObserver?.disconnect();
      const line = phone.matches ? 46 : 86;
      sectionObserver = new IntersectionObserver(
        (entries) => {
          if (suppressed) return;
          const hit = entries.find((entry) => entry.isIntersecting);
          if (!hit) return; // between sections or past the last one: keep the current selection
          const id = hit.target.id;
          select(navLinks.some((link) => link.dataset.navSection === id) ? id : null);
        },
        { rootMargin: `-${line}px 0px -${window.innerHeight - line - 1}px 0px` },
      );
      for (const section of sections) sectionObserver.observe(section);
    }
    observeSections();
    window.addEventListener('resize', observeSections);
    phone.addEventListener('change', observeSections);

    function release() {
      suppressed = false;
      window.clearTimeout(releaseTimer);
    }

    for (const link of navLinks) {
      link.addEventListener('click', () => {
        const id = link.dataset.navSection;
        if (!id) return;
        select(id);
        suppressed = true;
        window.clearTimeout(releaseTimer);
        releaseTimer = window.setTimeout(release, reduced.matches ? 0 : 1000);
        window.addEventListener('scrollend', release, { once: true });
      });
    }
  </script>
)}
```

- [ ] **Step 1: Write the checker**

`site/scripts/check-links.mjs`:

```js
import { existsSync, readFileSync, readdirSync, statSync } from 'node:fs';
import { join, relative } from 'node:path';

const dist = new URL('../dist/', import.meta.url).pathname;
const base = (process.env.BASE_PATH ?? '/').replace(/\/?$/, '/');

function* htmlFiles(dir) {
  for (const name of readdirSync(dir)) {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) yield* htmlFiles(path);
    else if (name.endsWith('.html')) yield path;
  }
}

function resolves(url) {
  let path = url.split('#')[0].split('?')[0];
  if (path === '') return true;
  if (!path.startsWith(base)) return false;
  path = decodeURI(path.slice(base.length));
  const target = join(dist, path);
  return (
    (existsSync(target) && statSync(target).isFile()) ||
    existsSync(join(target, 'index.html')) ||
    existsSync(`${target}.html`)
  );
}

const broken = [];
for (const file of htmlFiles(dist)) {
  const html = readFileSync(file, 'utf8');
  for (const [, url] of html.matchAll(/\s(?:href|src)="([^"]+)"/g)) {
    if (!url.startsWith('/') || url.startsWith('//')) continue;
    if (!resolves(url)) broken.push(`${relative(dist, file)} → ${url}`);
  }
}

if (broken.length > 0) {
  console.error(`Broken internal links (base ${base}):\n${broken.join('\n')}`);
  process.exit(1);
}
console.log(`All internal links resolve (base ${base}).`);
```
